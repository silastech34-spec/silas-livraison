name: Bootstrap Flutter Project

# Se lance manuellement, une seule fois, pour générer la structure du projet
on:
  workflow_dispatch:

jobs:
  bootstrap:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Flutter
        uses: subosito/flutter-action@v2
        with:
          flutter-version: '3.24.0'
          channel: 'stable'

      - name: Generate Flutter project in place
        run: |
          flutter create --org com.silaslivraison --project-name silas_livraison .

      - name: Add core dependencies
        run: |
          flutter pub add supabase_flutter go_router intl

      - name: Commit and push generated project
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add .
          git commit -m "Bootstrap: flutter create + dépendances de base"
          git push
